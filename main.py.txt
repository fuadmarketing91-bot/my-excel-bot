bot = telebot.TeleBot(TOKEN)

# В облаке Render файлы будут обрабатываться в изолированной папке /tmp
DOWNLOAD_DIR = "/tmp/processed_files"
os.makedirs(DOWNLOAD_DIR, exist_ok=True)

@bot.message_handler(commands=['start'])
def start_command(message):
    bot.send_message(
        message.chat.id, 
        "Привет! Я облачный бот-сегментатор базы.\n\n"
        "📁 **Отправьте мне ваш файл Excel (.xlsx)**, и я автоматически разделю его на сегменты по вкладкам:\n"
        "• **CL** (9-30 дней)\n"
        "• **CL2** (31-90 дней)\n"
        "• **CL3** (91-365 дней)"
    )

@bot.message_handler(content_types=['document'])
def handle_excel_file(message):
    try:
        file_name = message.document.file_name
        
        # Проверяем расширение файла
        if not file_name.endswith('.xlsx'):
            bot.reply_to(message, "❌ Ошибка! Нужен файл Excel с расширением .xlsx")
            return

        bot.reply_to(message, "⏳ Файл получен. Считаю дни и распределяю по сегментам...")

        # Скачиваем файл в изолированный контейнер
        file_info = bot.get_file(message.document.file_id)
        downloaded_file = bot.download_file(file_info.file_path)
        
        input_path = os.path.join(DOWNLOAD_DIR, file_name)
        with open(input_path, 'wb') as new_file:
            new_file.write(downloaded_file)

        # Читаем таблицу
        df = pd.read_excel(input_path)

        # Проверяем, есть ли нужная колонка
        if 'segment_days' not in df.columns:
            bot.send_message(
                message.chat.id, 
                f"❌ Ошибка: В таблице не найдена колонка `segment_days`.\nДоступные колонки: {', '.join(df.columns)}"
            )
            os.remove(input_path)
            return

        # Конвертируем данные в числа
        df['segment_days'] = pd.to_numeric(df['segment_days'], errors='coerce')

        # Фильтруем данные по вашим диапазонам
        cl_df = df[(df['segment_days'] >= 9) & (df['segment_days'] <= 30)]
        cl2_df = df[(df['segment_days'] >= 31) & (df['segment_days'] <= 90)]
        cl3_df = df[(df['segment_days'] >= 91) & (df['segment_days'] <= 365)]

        output_filename = f"сегментированный_{file_name}"
        output_path = os.path.join(DOWNLOAD_DIR, output_filename)

        # Записываем новые вкладки
        with pd.ExcelWriter(output_path, engine='openpyxl') as writer:
            df.to_excel(writer, sheet_name='Вся база', index=False)
            cl_df.to_excel(writer, sheet_name='CL (9-30d)', index=False)
            cl2_df.to_excel(writer, sheet_name='CL2 (31-90d)', index=False)
            cl3_df.to_excel(writer, sheet_name='CL3 (91-365d)', index=False)

        # Отправляем готовый файл обратно вам в чат
        with open(output_path, 'rb') as result_file:
            bot.send_document(
                message.chat.id, 
                result_file, 
                caption="🎉 Сегментация завершена! Все данные разложены по вкладкам согласно дням активности."
            )

        # БЕЗОПАСНОСТЬ: Полностью и безвозвратно удаляем файлы из памяти сервера
        os.remove(input_path)
        os.remove(output_path)

    except Exception as e:
        bot.send_message(message.chat.id, f"❌ Произошла ошибка при обработке файла: {e}")

if __name__ == '__main__':
    print("🤖 Облачный бот успешно запущен...")
    bot.infinity_polling()
